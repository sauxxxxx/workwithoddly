async(page)=>{
 await page.goto('http://localhost:3210/');await page.setViewportSize({width:1440,height:900});await page.evaluate(()=>document.fonts.ready);await page.waitForTimeout(1000);await page.evaluate(()=>{document.documentElement.style.scrollBehavior='auto';history.scrollRestoration='manual';scrollTo({top:0,behavior:'instant'});});await page.waitForTimeout(800);
 await page.screenshot({path:'output/perspective/hero.png'});
 await page.evaluate(()=>{document.querySelector('main').style.visibility='hidden';document.querySelector('[data-live-stage]').style.visibility='hidden';document.querySelector('header').style.visibility='hidden';});
 await page.screenshot({path:'public/art/studio-room.webp'});
 await page.evaluate(()=>{document.querySelector('main').style.visibility='';document.querySelector('[data-live-stage]').style.visibility='';document.querySelector('header').style.visibility='';});
 for(const [name,p] of [['studio',0],['transition',.46],['project',.86],['reverse',.36]]){
  await page.evaluate(p=>{const el=document.getElementById('studio');scrollTo({top:scrollY+el.getBoundingClientRect().top+(el.offsetHeight-innerHeight)*p,behavior:'instant'});},p);await page.waitForTimeout(800);await page.screenshot({path:'output/perspective/'+name+'.png'});
 }

}
