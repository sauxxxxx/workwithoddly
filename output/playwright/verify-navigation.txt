async (page) => {
  const errors=[];
  page.on('pageerror',error=>errors.push(error.message));
  const failed=[];
  page.on('response',response=>{if(response.status()>=400)failed.push({status:response.status(),url:response.url()});});
  await page.setViewportSize({width:1440,height:900});
  await page.goto('http://127.0.0.1:3214/');
  await page.evaluate(()=>document.documentElement.style.scrollBehavior='auto');
  await page.locator('#more-work').scrollIntoViewIfNeeded();
  await page.waitForTimeout(1000);
  const accounting=page.locator('#more-work ol a').filter({hasText:'Joyno Accounting'});
  await accounting.click();
  await page.waitForURL('**/work/joyno-accounting/');
  await page.getByRole('link',{name:'ODDLY — home',exact:true}).click();
  await page.waitForURL('http://127.0.0.1:3214/');
  await page.locator('#work').scrollIntoViewIfNeeded();
  await page.getByRole('link',{name:'Explore the project',exact:true}).click();
  await page.waitForURL('**/work/the-beach-park-hadsan/');
  const mobile=await page.context().browser().newContext({viewport:{width:390,height:844},isMobile:true,hasTouch:true});
  const touch=await mobile.newPage();
  await touch.goto('http://127.0.0.1:3214/');
  await touch.locator('#automation').scrollIntoViewIfNeeded();
  await touch.waitForTimeout(450);
  const native=await touch.evaluate(()=>matchMedia('(pointer:coarse)').matches&&!document.documentElement.classList.contains('lenis'));
  if(!native)throw new Error('Touch scrolling should be native');
  const hidden=await touch.locator('[data-live-stage]').getAttribute('data-hidden');
  if(hidden!=='true')throw new Error('Mobile WebGL should pause after the studio');
  await touch.getByRole('button',{name:'Preview CasaToon',exact:true}).tap();
  const expanded=await touch.getByRole('button',{name:'Preview CasaToon',exact:true}).getAttribute('aria-expanded');
  if(expanded!=='true')throw new Error('Touch preview should expand');
  await mobile.close();
  await page.goto('http://127.0.0.1:3214/');
  await page.waitForTimeout(600);
  await page.evaluate(()=>document.querySelector('canvas')?.getContext('webgl2')?.getExtension('WEBGL_lose_context')?.loseContext());
  await page.waitForTimeout(400);
  const fallback=await page.locator('img[src="/stills/opening-desktop.webp"]').isVisible();
  if(!fallback)throw new Error('Lost WebGL context should switch to the composed still');
  if(errors.length||failed.length)throw new Error(JSON.stringify({errors,failed}));
  await page.evaluate(data=>window.navigationVerification=data,{errors,failed,nativeTouch:native,mobileStageHidden:hidden,contextLossFallback:fallback});
}
